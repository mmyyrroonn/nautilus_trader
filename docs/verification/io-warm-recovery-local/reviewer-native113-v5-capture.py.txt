import datetime
import difflib
import hashlib
import json
import pathlib
import sys

root = pathlib.Path("E:/persarb/_tmp/entropy113-validation")
v4 = json.loads((root / "native113-frozen-source-v4.json").read_text())
manifest = root / "native113-frozen-source-v5.json"
v5 = json.loads(manifest.read_text())
old = {i["path"]: i for g in ("native", "app") for i in v4[g]["files"]}
identities = []
diff = []
changed = []
for group in ("native", "app"):
    base = pathlib.Path(v5[group]["root"])
    for entry in v5[group]["files"]:
        source = pathlib.Path(entry["path"])
        data = source.read_bytes()
        actual = hashlib.sha256(data).hexdigest()
        identities.append({"path": str(source), "size": len(data), "sha256": actual, "manifest_sha256": entry["sha256"], "matches": actual == entry["sha256"]})
        if old[str(source)]["sha256"] == entry["sha256"]:
            continue
        rel = source.relative_to(base)
        before = pathlib.Path(v4["snapshot_directory"]) / group / rel
        after = pathlib.Path(v5["snapshot_directory"]) / group / rel
        changed.append(str(source))
        diff.extend(difflib.unified_diff(before.read_text().splitlines(), after.read_text().splitlines(), fromfile=str(before), tofile=str(after), lineterm=""))
result = {"utc": datetime.datetime.now(datetime.timezone.utc).isoformat(), "argv": sys.argv, "python": sys.executable, "script_sha256": hashlib.sha256(pathlib.Path(__file__).read_bytes()).hexdigest(), "manifest": str(manifest), "manifest_sha256": hashlib.sha256(manifest.read_bytes()).hexdigest(), "files": identities, "v4_to_v5_changed": changed, "all_match": all(row["matches"] for row in identities), "independent_tests_run": 0}
out = root / f"reviewer-native113-source-v5-{sys.argv[1]}-identities.json"
out.write_text(json.dumps(result, ensure_ascii=False, indent=2) + "\n")
(root / "reviewer-native113-v4-v5-style-diff.txt").write_text("\n".join(diff) + "\n")
print(json.dumps({"utc": result["utc"], "report": str(out), "files": len(identities), "all_match": result["all_match"], "changed": changed}))
print("\n".join(diff))
