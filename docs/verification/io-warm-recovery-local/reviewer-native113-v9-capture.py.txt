import datetime
import difflib
import hashlib
import json
import pathlib
import sys

root = pathlib.Path("E:/persarb/_tmp/entropy113-validation")
manifest = root / "native113-frozen-source-v9.json"
previous = json.loads((root / "native113-frozen-source-v8.json").read_text())
current = json.loads(manifest.read_text())
old = {i["path"]: i for g in ("native", "app") for i in previous[g]["files"]}
identities, diff, changed = [], [], []
for group in ("native", "app"):
    base = pathlib.Path(current[group]["root"])
    for entry in current[group]["files"]:
        path = pathlib.Path(entry["path"])
        data = path.read_bytes()
        actual = hashlib.sha256(data).hexdigest()
        identities.append({"path": str(path), "size": len(data), "sha256": actual, "manifest_sha256": entry["sha256"], "matches": actual == entry["sha256"]})
        if old.get(str(path), {}).get("sha256") == entry["sha256"]:
            continue
        changed.append(str(path))
        rel = path.relative_to(base)
        before = pathlib.Path(previous["snapshot_directory"]) / group / rel
        after = pathlib.Path(current["snapshot_directory"]) / group / rel
        diff.extend(difflib.unified_diff((before.read_text() if before.exists() else "").splitlines(), after.read_text().splitlines(), fromfile=str(before), tofile=str(after), lineterm=""))
result = {"utc": datetime.datetime.now(datetime.timezone.utc).isoformat(), "argv": sys.argv, "python": sys.executable, "script_sha256": hashlib.sha256(pathlib.Path(__file__).read_bytes()).hexdigest(), "manifest": str(manifest), "manifest_sha256": hashlib.sha256(manifest.read_bytes()).hexdigest(), "files": identities, "v8_to_v9_changed": changed, "all_match": all(row["matches"] for row in identities), "independent_tests_run": 0}
out = root / f"reviewer-native113-source-v9-{sys.argv[1]}-identities.json"
out.write_text(json.dumps(result, ensure_ascii=False, indent=2) + "\n")
(root / "reviewer-native113-v8-v9-fixture-diff.txt").write_text("\n".join(diff) + "\n")
print(json.dumps({"utc": result["utc"], "report": str(out), "files": len(identities), "all_match": result["all_match"], "changed": changed}))
print("\n".join(diff))
