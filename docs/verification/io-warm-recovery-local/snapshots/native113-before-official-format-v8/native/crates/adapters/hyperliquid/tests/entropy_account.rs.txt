#[derive(Clone, Debug)]

    Close,
    Frame(Value),
    RawFrame(String),

struct PeerData {
    mode: Value,
    legacy: Value,
    role: Value,
    collateral: usize,
    token_id: String,
    io: Value,
    ws_financial_override: Option<Value>,
    default: Value,
    spot: Value,
    time_offset_ms: u64,
    missing_time: bool,
    future_time_ms: u64,
    io_response_delay_ms: u64,
    failed_endpoint: Option<String>,
    ws_enabled: bool,
    acknowledge: bool,
    requests: Vec<Value>,
    http_observations: Vec<Value>,
    ws_observations: Vec<Value>,
    subscriptions: Vec<Value>,
    execution: Option<execution::PeerExecution>,
    meta_override: Option<Value>,
}

#[derive(Clone)]
struct PeerState {
    data: Arc<Mutex<PeerData>>,
    instructions: tokio::sync::broadcast::Sender<PeerInstruction>,
    connections: Arc<AtomicUsize>,
    active: Arc<AtomicUsize>,
    writes: Arc<AtomicUsize>,
}

impl PeerState {
    fn new(io: Value) -> Self {
        let (instructions, _) = tokio::sync::broadcast::channel(32);
        Self {
            data: Arc::new(Mutex::new(PeerData {
                mode: json!("disabled"),
                legacy: json!(false),
                role: json!({"role":"user"}),
                collateral: 0,
                token_id: USDC_ID.to_string(),
                io,
                ws_financial_override: None,
                default: clearinghouse("90000", "90000", "0", "80000", false),
                spot: json!({"balances":[
                    {"coin":"USDC", "token":0, "total":"95000", "hold":"500", "entryNtl":"0"}
                ]}),
                time_offset_ms: 0,
                missing_time: false,
                future_time_ms: 0,
                io_response_delay_ms: 0,
                failed_endpoint: None,
                ws_enabled: true,
                acknowledge: true,
                requests: Vec::new(),
                http_observations: Vec::new(),
                ws_observations: Vec::new(),
                subscriptions: Vec::new(),
                execution: None,
                meta_override: None,
            })),
            instructions,
            connections: Arc::new(AtomicUsize::new(0)),
            active: Arc::new(AtomicUsize::new(0)),
            writes: Arc::new(AtomicUsize::new(0)),
        }
    }

    fn meta(&self) -> Value {
        let data = self.data.lock();
        if let Some(meta) = &data.meta_override {
            return meta.clone();
        }
        json!({"universe":[{"name":"io:SNDK", "szDecimals":4, "maxLeverage":10,
            "onlyIsolated":true, "marginMode":"strictIsolated"}], "collateralToken":data.collateral})
    }

    fn clearinghouse_frame(&self) -> Value {
        let data = self.data.lock();
        json!({"channel":"clearinghouseState", "data":{
            "user":USER, "dex":"io", "clearinghouseState":data.ws_financial_override.as_ref().unwrap_or(&data.io).clone()
        }})
    }
}

struct Peer {
    state: PeerState,
    addr: SocketAddr,
    server: JoinHandle<()>,
}

impl Peer {
    async fn start(io: Value) -> Self {
        let state = PeerState::new(io);
        let listener = tokio::net::TcpListener::bind("127.0.0.1:0").await.unwrap();
        let addr = listener.local_addr().unwrap();
        let app = Router::new()
            .route("/info", post(info))
            .route("/exchange", post(exchange))
            .route("/ws", get(upgrade))
            .with_state(state.clone());
        let server = tokio::spawn(async move { axum::serve(listener, app).await.unwrap() });
        Self {
            state,
            addr,
            server,
        }
    }
}

impl Drop for Peer {
    fn drop(&mut self) {
        self.server.abort();
    }
}

fn now_ms() -> u64 {
    u64::try_from(
        SystemTime::now()
            .duration_since(UNIX_EPOCH)
            .unwrap()
            .as_millis(),
    )
    .unwrap()
}

fn clearinghouse(raw: &str, equity: &str, used: &str, withdrawable: &str, position: bool) -> Value {
    let positions = if position {
        vec![json!({"type":"oneWay", "position":{
            "coin":"io:SNDK", "szi":"0.5", "marginUsed":used,
            "leverage":{"type":"isolated", "value":2}, "cumFunding":{
                "allTime":"0", "sinceOpen":"0", "sinceChange":"0"},
            "entryPx":"100", "liquidationPx":null, "maxLeverage":10,
            "positionValue":"50", "returnOnEquity":"0", "unrealizedPnl":"-4.625"
        }})]
    } else {
        vec![]
    };
    json!({
        "marginSummary":{"accountValue":equity, "totalNtlPos":if position {"50"} else {"0"},
            "totalRawUsd":raw, "totalMarginUsed":used},
        "crossMarginSummary":{"accountValue":"0", "totalNtlPos":"0",
            "totalRawUsd":"0", "totalMarginUsed":"0"},
        "crossMaintenanceMarginUsed":"0", "withdrawable":withdrawable, "assetPositions":positions
    })
}

fn record_http_response(
    state: &PeerState,
    request: &Value,
    started: u64,
    status: u16,
    raw: &str,
) {
    let mut data = state.data.lock();
    if data
        .execution
        .as_ref()
        .is_some_and(|execution| execution.record_http)
    {
        data.http_observations
            .push(json!({"request":request,"started_ms":started,
            "completed_ms":now_ms(),"status":status,"raw_body":raw}));
    }
}

async fn info(State(state): State<PeerState>, Json(request): Json<Value>) -> Response {
    let started = now_ms();
    let kind = request["type"].as_str().unwrap_or("").to_string();
    let failed = {
        let mut data = state.data.lock();
        data.requests.push(request.clone());
        data.failed_endpoint.as_deref() == Some(kind.as_str())
    };
    if failed {
        record_http_response(&state, &request, started, 503, "synthetic outage");
        return (
            axum::http::StatusCode::SERVICE_UNAVAILABLE,
            "synthetic outage",
        )
            .into_response();
    }
    if kind == "clearinghouseState" && request["dex"] == "io" {
        let delay = state.data.lock().io_response_delay_ms;
        if delay > 0 {
            tokio::time::sleep(Duration::from_millis(delay)).await;
        }
    }
    if let Some((delay_ms, raw_body)) = execution::economic_info_response(&state, &request) {
        if delay_ms > 0 {
            tokio::time::sleep(Duration::from_millis(delay_ms)).await;
        }
        record_http_response(&state, &request, started, 200, &raw_body);
        return (
            [(axum::http::header::CONTENT_TYPE, "application/json")],
            raw_body,
        )
            .into_response();
    }
    if let Some((delay_ms, raw_body)) = execution::startup_info_response(&state, &request) {
        if delay_ms > 0 {
            tokio::time::sleep(Duration::from_millis(delay_ms)).await;
        }
        record_http_response(&state, &request, started, 200, &raw_body);
        return (
            [(axum::http::header::CONTENT_TYPE, "application/json")],
            raw_body,
        )
            .into_response();
    }
    let response = match kind.as_str() {
        "meta" => state.meta(),
        "allPerpMetas" => json!([{"universe":[], "collateralToken":0}, state.meta()]),
        "perpDexs" => json!([null, {"name":"io", "fullName":"Synthetic Entropy"}]),
        "spotMeta" => {
            let data = state.data.lock();
            json!({"universe":[], "tokens":[{"name":"USDC", "index":0,
                "szDecimals":8, "weiDecimals":8, "tokenId":data.token_id,
                "isCanonical":true, "fullName":"USD Coin"}]})
        }
        "spotMetaAndAssetCtxs" => json!([{"universe":[], "tokens":[]}, []]),
        "userRole" => state.data.lock().role.clone(),
        "userAbstraction" => state.data.lock().mode.clone(),
        "userDexAbstraction" => state.data.lock().legacy.clone(),
        "clearinghouseState" => {
            let data = state.data.lock();
            let mut response = if request["dex"] == "io" {
                data.io.clone()
            } else {
                data.default.clone()
            };
            if !data.missing_time {
                response["time"] = json!(
                    now_ms()
                        .saturating_sub(data.time_offset_ms)
                        .saturating_add(data.future_time_ms)
                );
            }
            response
        }
        "spotClearinghouseState" => state.data.lock().spot.clone(),
        "userFees" => json!({"userCrossRate":"0.00045", "userAddRate":"0.00015"}),
        "userFills" | "historicalOrders" | "openOrders" | "frontendOpenOrders" => {
            execution::info_response(&state, &request).unwrap_or_else(|| json!([]))
        }
        _ => match execution::info_response(&state, &request) {
            Some(response) => response,
            None => {
                record_http_response(
                    &state,
                    &request,
                    started,
                    400,
                    "unexpected info type",
                );
                return (axum::http::StatusCode::BAD_REQUEST, "unexpected info type")
                    .into_response();
            }
        },
    };
    record_http_response(&state, &request, started, 200, &response.to_string());
    Json(response).into_response()
}

async fn exchange(State(state): State<PeerState>) -> Response {
    state.writes.fetch_add(1, Ordering::SeqCst);
    (
        axum::http::StatusCode::FORBIDDEN,
        "account tests never permit writes",
    )
        .into_response()
}

async fn upgrade(ws: WebSocketUpgrade, State(state): State<PeerState>) -> Response {
    ws.on_upgrade(move |socket| stream(socket, state))
}

struct ActiveSocket(Arc<AtomicUsize>);
impl Drop for ActiveSocket {
    fn drop(&mut self) {
        self.0.fetch_sub(1, Ordering::SeqCst);
    }
}

async fn stream(mut socket: WebSocket, state: PeerState) {
    state.connections.fetch_add(1, Ordering::SeqCst);
    state.active.fetch_add(1, Ordering::SeqCst);
    let _active = ActiveSocket(state.active.clone());
    let mut instructions = state.instructions.subscribe();
    let mut subscribed = false;
    // Execution workflows explicitly publish each changed private fact and pace
    // heavy REST verification under the real limiter. Keep the original account
    // fixtures' 60ms stream, with a slower heartbeat for these finite workflows.
    let heartbeat = if state.data.lock().execution.is_some() {
        Duration::from_secs(5)
    } else {
        Duration::from_millis(60)
    };
    let mut interval = tokio::time::interval(heartbeat);
    loop {
        let outgoing = tokio::select! {
            incoming = socket.recv() => {
                let Some(Ok(message)) = incoming else { break };
                match message {
                    Message::Text(text) => {
                        let Ok(value) = serde_json::from_str::<Value>(&text) else { break };
                        match value["method"].as_str() {
                            Some("subscribe") => {
                                let subscription = value["subscription"].clone();
                                subscribed |= subscription["type"] == "clearinghouseState";
                                if subscription["type"] == "clearinghouseState" {
                                    interval.reset_immediately();
                                }
                                state.data.lock().subscriptions.push(subscription.clone());
                                if state.data.lock().acknowledge {
                                    Some(Message::Text(json!({"channel":"subscriptionResponse", "data":{
                                        "method":"subscribe", "subscription":subscription
                                    }}).to_string().into()))
                                } else { None }
                            }
                            Some("ping") => Some(Message::Text(json!({"channel":"pong"}).to_string().into())),
                            Some("unsubscribe") => None,
                            Some("post") => {
                                state.writes.fetch_add(1, Ordering::SeqCst);
                                execution::post_response(&state, &value)
                                    .map(|reply| Message::Text(reply.to_string().into()))
                            }
                            _ => None,
                        }
                    }
                    Message::Ping(bytes) => Some(Message::Pong(bytes)),
                    Message::Close(_) => break,
                    _ => None,
                }
            }
            command = instructions.recv() => match command {
                Ok(PeerInstruction::Close) => { let _ = socket.send(Message::Close(None)).await; break; }
                Ok(PeerInstruction::Frame(value)) => Some(Message::Text(value.to_string().into())),
                Ok(PeerInstruction::RawFrame(text)) => Some(Message::Text(text.into())),
                Err(_) => None,
            },
            _ = interval.tick() => {
                if subscribed && state.data.lock().ws_enabled {
                    Some(Message::Text(state.clearinghouse_frame().to_string().into()))
                } else { None }
            }
        };
        if let Some(message) = outgoing {
            let text = match &message {
                Message::Text(text) => Some(text.to_string()),
                _ => None,
            };
            let sent = socket.send(message).await;
            {
                let mut data = state.data.lock();
                if data
                    .execution
                    .as_ref()
                    .is_some_and(|execution| execution.record_http)
                    && let Some(text) = text
                {
                    data.ws_observations.push(json!({"completed_ms":now_ms(),
                        "raw_body":text,"delivered":sent.is_ok()}));
                }
            }
            if sent.is_err() {
                break;
            }
        }
    }
}

fn make_client(
    peer: &Peer,
    max_age_ms: u64,
) -> (
    HyperliquidExecutionClient,
    UnboundedReceiver<ExecutionEvent>,
    Rc<RefCell<Cache>>,
) {
    let cache = Rc::new(RefCell::new(Cache::default()));
    let core = ExecutionClientCore::new(
        TraderId::from("TESTER-001"),
        *HYPERLIQUID_CLIENT_ID,
        *HYPERLIQUID_VENUE,
        OmsType::Netting,
        AccountId::from("HYPERLIQUID-ENTROPY"),
        AccountType::Margin,
        None,
        cache.clone(),
    );
    let config = execution_config(peer, max_age_ms);
    let (sender, receiver) = tokio::sync::mpsc::unbounded_channel();
    set_exec_event_sender(sender);
    let mut client = HyperliquidExecutionClient::new(core, config).unwrap();
    client.start().unwrap();
    (client, receiver, cache)
}

fn execution_config(peer: &Peer, max_age_ms: u64) -> HyperliquidExecutionClientConfig {
    HyperliquidExecutionClientConfig {
        account_id: AccountId::from("HYPERLIQUID-ENTROPY"),
        private_key: Some(KEY.to_string()),
        account_address: Some(USER.to_string()),
        environment: HyperliquidEnvironment::Testnet,
        base_url_http: Some(format!("http://{}/info", peer.addr)),
        base_url_ws: Some(format!("ws://{}/ws", peer.addr)),
        base_url_exchange: Some(format!("http://{}/exchange", peer.addr)),
        account_dex: Some("io".to_string()),
        account_snapshot_max_age_ms: max_age_ms,
        include_builder_attribution: false,
        max_retries: 0,
        http_timeout_secs: 2,
        ws_post_timeout_secs: 1,
        transport_backend: TransportBackend::Tungstenite,
        ..Default::default()
    }
}

fn snapshot(client: &HyperliquidExecutionClient) -> Value {
    client
        .account_scope_snapshot_json()
        .unwrap()
        .map(|value| serde_json::from_str(&value).unwrap())
        .unwrap_or_else(|| json!({"trusted":false, "flat":null}))
}

async fn connect_register(
    client: &mut HyperliquidExecutionClient,
    receiver: &mut UnboundedReceiver<ExecutionEvent>,
    cache: &Rc<RefCell<Cache>>,
) -> AccountState {
    let account = connect_collect(client, receiver, cache).await;
    wait_trusted(client, true).await;
    account
}

async fn connect_collect(
    client: &mut HyperliquidExecutionClient,
    receiver: &mut UnboundedReceiver<ExecutionEvent>,
    cache: &Rc<RefCell<Cache>>,
) -> AccountState {
    let (connected, account) = connect_attempt(client, receiver, cache).await;
    assert!(
        connected.is_ok(),
        "{connected:?}; scope={}",
        snapshot(client)
    );
    account.unwrap()
}

async fn connect_attempt(
    client: &mut HyperliquidExecutionClient,
    receiver: &mut UnboundedReceiver<ExecutionEvent>,
    cache: &Rc<RefCell<Cache>>,
) -> (anyhow::Result<()>, anyhow::Result<AccountState>) {
    tokio::join!(client.connect(), register_account(receiver, cache))
}

async fn register_account(
    receiver: &mut UnboundedReceiver<ExecutionEvent>,
    cache: &Rc<RefCell<Cache>>,
) -> anyhow::Result<AccountState> {
    let event = tokio::time::timeout(Duration::from_secs(3), receiver.recv())
        .await
        .map_err(|_| anyhow::anyhow!("native account emission timed out"))?
        .ok_or_else(|| anyhow::anyhow!("native execution event channel closed"))?;
    let ExecutionEvent::Account(account) = event else {
        panic!("expected native account state")
    };
    cache
        .borrow_mut()
        .add_account(AccountAny::Margin(MarginAccount::new(
            account.clone(),
            true,
        )))
        .unwrap();
    Ok(account)
}

async fn wait_trusted(client: &HyperliquidExecutionClient, trusted: bool) {
    tokio::time::timeout(Duration::from_secs(3), async {
        loop {
            if snapshot(client)["trusted"] == trusted {
                break;
            }
            tokio::time::sleep(Duration::from_millis(5)).await;
        }
    })
    .await
    .unwrap_or_else(|error| {
        panic!(
            "scope trust did not reach {trusted}: {error}; scope={}",
            snapshot(client)
        )
    });
}

fn query(client: &HyperliquidExecutionClient) {
    client
        .query_account(QueryAccount::new(
            TraderId::from("TESTER-001"),
            Some(*HYPERLIQUID_CLIENT_ID),
            AccountId::from("HYPERLIQUID-ENTROPY"),
            UUID4::new(),
            UnixNanos::default(),
            None,
            None,
        ))
        .unwrap();
}

async fn stop(client: &mut dyn ExecutionClient, peer: &Peer) {
    client.disconnect().await.unwrap();
    assert!(!client.is_connected());
    assert_eq!(peer.state.writes.load(Ordering::SeqCst), 0);
    tokio::time::timeout(Duration::from_secs(2), async {
        while peer.state.active.load(Ordering::SeqCst) > 0 {
            tokio::task::yield_now().await;
        }
    })
    .await
    .unwrap();
}

#[rstest]
#[tokio::test]
async fn io_zero_balance_never_inherits_default_or_spot_wealth() {
    let peer = Peer::start(clearinghouse("0", "0", "0", "0", false)).await;
    let (mut client, mut receiver, cache) = make_client(&peer, 30_000);
    let account = connect_register(&mut client, &mut receiver, &cache).await;
    assert_eq!(account.balances.len(), 1);
    assert_eq!(account.balances[0].total, Money::from("0 USDC"));
    let scope = snapshot(&client);
    assert_eq!(scope["dex"], "io");
    assert_eq!(scope["address"], USER);
    assert_eq!(scope["flat"], true);
    assert_eq!(scope["balance"], "0");
    assert!(scope["ws_source_time_ms"].is_null());
    let requests = peer.state.data.lock().requests.clone();
    assert!(
        requests
            .iter()
            .any(|value| value["type"] == "clearinghouseState"
                && value["dex"] == "io"
                && value["user"] == USER)
    );
    assert!(
        !requests
            .iter()
            .any(|value| value["type"] == "spotClearinghouseState"
                || (value["type"] == "clearinghouseState" && value["dex"] != "io"))
    );
    for subscription in &peer.state.data.lock().subscriptions {
        assert_eq!(subscription["user"], USER);
        if subscription["type"] == "clearinghouseState" {
            assert_eq!(subscription["dex"], "io");
        }
    }
    stop(&mut client, &peer).await;
}

#[rstest]
#[tokio::test]
async fn isolated_io_equity_used_free_and_withdrawable_remain_distinct() {
    let peer = Peer::start(clearinghouse("42.125", "37.5", "9.25", "11", true)).await;
    peer.state.data.lock().default = clearinghouse("0", "0", "0", "0", false);
    let (mut client, mut receiver, cache) = make_client(&peer, 30_000);
    let account = connect_register(&mut client, &mut receiver, &cache).await;
    let scope = snapshot(&client);
    assert_eq!(scope["balance"], "42.125");
    assert_eq!(scope["equity"], "37.5");
    assert_eq!(scope["used"], "9.25");
    assert_eq!(scope["free"], "28.25");
    assert_eq!(scope["withdrawable"], "11");
    assert_eq!(scope["flat"], false);
    assert_eq!(scope["positions"][0]["coin"], "io:SNDK");
    assert_eq!(account.balances[0].total, Money::from("37.5 USDC"));
    assert_eq!(account.balances[0].locked, Money::from("9.25 USDC"));
    assert_eq!(account.balances[0].free, Money::from("28.25 USDC"));
    assert!(account.margins.is_empty());
    query(&client);
    tokio::time::timeout(Duration::from_secs(2), async {
        while !client.pending_tasks_all_finished() {
            tokio::task::yield_now().await;
        }
    })
    .await
    .unwrap();
    wait_trusted(&client, true).await;
    assert_eq!(snapshot(&client)["withdrawable"], "11");
    stop(&mut client, &peer).await;
}

#[rstest]
#[case("unifiedAccount")]
#[case("portfolioMargin")]
#[case("dexAbstraction")]
#[case("default")]
#[case("unrecognized")]
#[tokio::test]
async fn unsupported_shared_and_unknown_modes_emit_no_io_account(#[case] mode: &str) {
    let peer = Peer::start(clearinghouse("0", "0", "0", "0", false)).await;
    peer.state.data.lock().mode = json!(mode);
    let (mut client, mut receiver, cache) = make_client(&peer, 30_000);
    assert!(client.connect().await.is_err());
    assert!(
        cache
            .borrow()
            .account(&AccountId::from("HYPERLIQUID-ENTROPY"))
            .is_none()
    );
