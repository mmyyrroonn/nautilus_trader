"""Actual installed Aster normal factory/LiveNode selected-scope read-only check."""

import argparse
import asyncio
import hashlib
import json
import sys
import time
import traceback
from datetime import timedelta
from pathlib import Path

APP = Path("E:/persarb/worktrees/entropy-execution-app")
sys.path.insert(0, str(APP / "src"))
sys.path.insert(0, str(APP / "tests"))
from entropy_execution_peer import ASTER_ID, KEY, USER, ExecutionPeer

from entropy_execution import verify_candidate


async def check(args):
    directory = Path(args.directory)
    directory.mkdir(parents=True, exist_ok=False)
    candidate = verify_candidate(args.wheel, args.sha256, args.provenance)
    manifest = {
        "candidate": candidate,
        "synthetic_numeric_loopback": True,
        "native_host_egress_isolation_verified": False,
        "orders_authorized": False,
        "reconciliation": True,
        "runner_sha256": hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
    }
    observations = []
    result = {
        "started": False,
        "stopped": False,
        "queries": 0,
        "shutdown_verified": False,
        "actual_selected_scope": None,
    }
    node = None
    failure = None
    peer = ExecutionPeer("ASTER")
    try:
        async with peer:
            from nautilus_trader.adapters.aster import (
                AsterDataClientConfig,
                AsterDataClientFactory,
                AsterEnvironment,
                AsterExecutionClientConfig,
                AsterExecutionClientFactory,
            )
            from nautilus_trader.adapters.binance import BinanceInstrumentProviderConfig
            from nautilus_trader.common import Environment, LoggerConfig, LogLevel
            from nautilus_trader.live import LiveExecutionEngineConfig, LiveNode
            from nautilus_trader.model import AccountId, ClientId, StrategyId, TraderId
            from nautilus_trader.trading import Strategy, StrategyConfig

            account = AccountId.from_str("ASTER-ENTROPY109")
            policy = {
                "instrument_ids": [ASTER_ID],
                "balance_asset": "USD1",
                "max_age_ms": 2000,
                "max_refresh_ms": 15000,
            }
            provider = BinanceInstrumentProviderConfig(
                load_all=False, load_ids=[ASTER_ID]
            )
            factory = AsterExecutionClientFactory()
            manifest.update(
                http_url=peer.http_url, ws_url=peer.ws_url, selected_scope_policy=policy
            )
            (directory / "manifest.json").write_text(
                json.dumps(manifest, indent=2), encoding="utf-8"
            )
            data = AsterDataClientConfig(
                environment=AsterEnvironment.TESTNET,
                instrument_provider=provider,
                base_url_http=peer.http_url,
                base_url_ws=peer.ws_url,
            )
            execution = AsterExecutionClientConfig(
                account_id=account,
                environment=AsterEnvironment.TESTNET,
                user_address=USER,
                signer_address=USER,
                signer_private_key=KEY,
                base_url_http=peer.http_url,
                base_url_ws=peer.ws_url,
                instrument_provider=provider,
                http_timeout_secs=2,
                ws_connect_timeout_secs=2,
                assume_one_way_mode_when_unconfirmed=False,
                selected_scope_policy_json=json.dumps(policy, separators=(",", ":")),
            )
            builder = LiveNode.builder(
                "ASTER-SELECTED-READONLY",
                TraderId.from_str("ASTER-109"),
                Environment.LIVE,
            )
            builder = (
                builder.with_logging(LoggerConfig(stdout_level=LogLevel.ERROR))
                .with_timeout_connection(5)
                .with_timeout_reconciliation(5)
                .with_timeout_portfolio(5)
                .with_timeout_disconnection_secs(5)
                .with_delay_post_stop_secs(0)
                .with_delay_shutdown_secs(0)
                .with_exec_engine_config(
                    LiveExecutionEngineConfig(
                        reconciliation=True,
                        filter_unclaimed_external_orders=False,
                        inflight_check_interval_ms=0,
                        open_check_interval_secs=None,
                        position_check_interval_secs=None,
                    )
                )
            )
            node = (
                builder.add_data_client("ASTER", AsterDataClientFactory(), data)
                .add_exec_client("ASTER", factory, execution)
                .build()
            )
            handle = node.handle()

            class ReadOnly(Strategy):
                def __init__(self):
                    super().__init__(
                        StrategyConfig(
                            strategy_id=StrategyId.from_str("ASTER-SCOPE-109"),
                            log_events=False,
                            log_commands=False,
                            manage_stop=False,
                        )
                    )
                    self.started_ns = None
                    self.prequery_generation = None

                def on_start(self):
                    result["started"] = True
                    self.started_ns = time.time_ns()
                    raw = factory.selected_scope_snapshot_json()
                    before = None if raw is None else json.loads(raw)
                    self.prequery_generation = (
                        None if before is None else before.get("generation")
                    )
                    result["prequery_scope"] = before
                    self.clock.set_timer(
                        "aster-proof-poll", timedelta(milliseconds=100)
                    )
                    self.query_account(account, client_id=ClientId.from_str("ASTER"))
                    result["queries"] += 1

                def on_time_event(self, event):
                    raw = factory.selected_scope_snapshot_json()
                    value = None if raw is None else json.loads(raw)
                    if not observations or observations[-1]["snapshot"] != value:
                        observations.append(
                            {"received_ns": time.time_ns(), "snapshot": value}
                        )
                    ready = bool(
                        value
                        and value.get("ready") is True
                        and value.get("trusted") is True
                    )
                    newer = value and (
                        self.prequery_generation is None
                        or value["generation"] > self.prequery_generation
                    )
                    if ready and newer:
                        result["actual_selected_scope"] = value
                        handle.stop()
                    elif time.time_ns() - self.started_ns > 5_000_000_000:
                        result["scope_timeout"] = True
                        handle.stop()

                def on_stop(self):
                    self.clock.cancel_timer("aster-proof-poll")
                    result["stopped"] = True

            node.add_strategy(ReadOnly())
            try:
                await asyncio.wait_for(node.run_async(), 18)
                result["shutdown_verified"] = (
                    result["stopped"] and not handle.is_running
                )
            finally:
                node.dispose()
                node = None
                result["post_shutdown_scope_json"] = (
                    factory.selected_scope_snapshot_json()
                )
    except (
        ValueError,
        RuntimeError,
        TypeError,
        KeyError,
        AttributeError,
        OSError,
        ArithmeticError,
        AssertionError,
    ) as exc:
        failure = f"{type(exc).__name__}: {exc}"
        traceback.print_exc()
    finally:
        if node is not None:
            node.dispose()
        (directory / "peer-evidence.json").write_text(
            json.dumps(
                {
                    "requests": peer.requests,
                    "commands": peer.commands,
                    "frames": peer.frames,
                    "errors": peer.errors,
                    "actions": peer.actions,
                    "fills": peer.fills,
                    "position": str(peer.position),
                    "active_sessions_after_teardown": len(peer.sessions),
                },
                indent=2,
            ),
            encoding="utf-8",
        )
        result.update(
            failure=failure,
            observations=observations,
            active_sessions_after_teardown=len(peer.sessions),
            native_host_egress_isolation_verified=False,
        )
        (directory / "result.json").write_text(
            json.dumps(result, indent=2), encoding="utf-8"
        )
    assert failure is None, failure
    assert result["started"] and result["stopped"] and result["shutdown_verified"], (
        result
    )
    value = result["actual_selected_scope"]
    assert value and value["ready"] is True and value["trusted"] is True, result
    assert value["account_id"] == "ASTER-ENTROPY109" and value["generation"] >= 1
    assert (
        value["stream_live"] is True
        and value["mode_verified"] is True
        and value["recovery_complete"] is True
    )
    assert value["positions"] == [
        {
            "instrument_id": ASTER_ID,
            "signed_quantity": "0",
            "source_update_time_ms": value["positions"][0]["source_update_time_ms"],
        }
    ]
    assert value["balance"]["asset"] == "USD1" and value["balance"]["free"] == "100"
    assert (
        value["whole_account_verified"] is False and value["funding_verified"] is False
    )
    assert not peer.sessions and not peer.errors and not peer.actions and not peer.fills
    print(
        json.dumps(
            {
                "actual_readonly_aster_selected_scope": "PASS",
                "result_path": str(directory / "result.json"),
            }
        ),
        flush=True,
    )


if __name__ == "__main__":
    parser = argparse.ArgumentParser()
    parser.add_argument("directory")
    parser.add_argument("wheel")
    parser.add_argument("sha256")
    parser.add_argument("provenance")
    asyncio.run(check(parser.parse_args()))
