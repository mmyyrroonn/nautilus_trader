import datetime as dt
import hashlib
import json
import subprocess
import sys
from pathlib import Path

repo = Path("E:/persarb/worktrees/entropy-aster-proof")
paths = sorted([
    *repo.glob("crates/adapters/aster/src/**/*.rs"),
    repo / "crates/adapters/binance/src/futures/websocket/streams/client.rs",
    repo / "crates/adapters/aster/tests/selected_scope.rs",
    repo / "crates/adapters/aster/tests/common/mod.rs",
    repo / "python/tests/unit/test_aster_selected_scope_config.py",
])
record = {
    "captured_utc": dt.datetime.now(dt.timezone.utc).isoformat(),
    "head": subprocess.check_output(["git", "rev-parse", "HEAD"], cwd=repo, text=True).strip(),
    "paths": {p.relative_to(repo).as_posix(): hashlib.sha256(p.read_bytes()).hexdigest() for p in paths},
    "note": "Exact scoped bytes, not a clean build/source fingerprint. Python-feature docstrings may be regenerated by the official generator.",
}
out = Path(__file__).parent / (sys.argv[1] + ".json")
out.write_text(json.dumps(record, indent=2) + "\n", encoding="utf-8")
print(json.dumps({"path": str(out), "count": len(paths)}))
