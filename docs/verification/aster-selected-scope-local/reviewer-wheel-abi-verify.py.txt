def digest_stream(stream):
    result = hashlib.sha256()
    while block := stream.read(1024 * 1024):
        result.update(block)
    return result.hexdigest()

def digest(path):
    with path.open('rb') as stream:
        return digest_stream(stream)

before = digest(wheel)
assert before == EXPECTED_WHEEL == manifest['wheel']['sha256']
assert wheel.stat().st_size == manifest['wheel']['size']
assert manifest['source_binding'] == 'verified'

            embedded = digest_stream(stream)
        actual = digest(path)

        records.append({'path': str(path), 'sha256': actual})

def prohibited(*args, **kwargs):
    raise AssertionError('Python environment/network lookup was attempted')

def audit(event, args):
    if event.startswith('socket.') or event.startswith('subprocess.'):
        raise AssertionError('Network/process creation was attempted: ' + event)
    if event == 'import' and str(args[0]).split('.')[0] == 'dotenv':
        raise AssertionError('dotenv import was attempted')

sys.addaudithook(audit)

def reject(name, *, policy_raw=raw, options=None):
    try:
        Config(**(values | (options or {})), selected_scope_policy_json=policy_raw)
    except ValueError:
        passed(name)
    else:
        raise AssertionError('Native constructor accepted forbidden policy: ' + name)
