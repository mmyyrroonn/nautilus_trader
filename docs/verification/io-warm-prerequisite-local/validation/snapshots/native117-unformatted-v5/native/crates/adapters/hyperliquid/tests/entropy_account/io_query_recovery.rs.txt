//! Ordinary bounded warm io QueryAccount through synthetic native HTTP/WS sources.

use std::time::Instant;

use super::*;

const SELECTED: &str = "io:SNDK-USD-PERP.HYPERLIQUID";

fn configure(config: &mut HyperliquidExecutionClientConfig, directory: &TempDir) {
    config.account_snapshot_max_age_ms = 2000;
    let mut selected = policy(directory);
    selected["action_timeout_ms"] = json!(2000);
    selected["recovery_timeout_ms"] = json!(3000);
    selected["metadata_max_age_ms"] = json!(30000);
    config.io_execution_policy_json = Some(selected.to_string());
}

async fn normal() -> Harness {
    Harness::with_configuration(
        "100",
        |peer| {
            peer.record_http = true;
        },
        configure,
    )
    .await
}

fn issue(harness: &Harness) {
    harness.client.query_account(command()).unwrap();
}

fn command() -> QueryAccount {
    QueryAccount::new(
        TraderId::from("TESTER-001"),
        Some(ClientId::from("HYPERLIQUID")),
        AccountId::from("HYPERLIQUID-ENTROPY"),
        UUID4::new(),
        UnixNanos::from(now_ms() * 1_000_000),
        None,
        None,
    )
}

async fn finished(harness: &Harness) -> Value {
    tokio::time::timeout(Duration::from_millis(3800), async {
        loop {
            harness.apply_time_events();
            let scope = harness.scope();
            let diagnostic = scope["diagnostic"].as_str().unwrap_or("").to_lowercase();
            if (scope["recovery_complete"] == true
                && scope["account"]["trusted"] == true
                && scope["query_in_flight"] != true)
                || (scope["query_in_flight"] != true
                    && !diagnostic.contains("pending")
                    && (diagnostic.contains("failed")
                        || diagnostic.contains("incomplete")
                        || diagnostic.contains("deadline exhausted")
                        || diagnostic.contains("superseded")))
            {
                if scope["recovery_complete"] != true {
                    let data = harness.peer.state.data.lock();
                    eprintln!(
                        "normal_query_unknown_scope={scope} actual_http={} actual_ws={}",
                        json!(data.http_observations),
                        json!(data.ws_observations)
                    );
                }
                return scope;
            }
            tokio::time::sleep(Duration::from_millis(5)).await;
        }
    })
    .await
    .unwrap_or_else(|_| {
        let data = harness.peer.state.data.lock();
        panic!(
            "ordinary query failed to settle finitely: scope={} actual_http={} actual_ws={}",
            harness.scope(),
            json!(data.http_observations),
            json!(data.ws_observations)
        )
    })
}

async fn await_request(harness: &Harness, kind: &str, after: usize) {
    tokio::time::timeout(Duration::from_secs(2), async {
        loop {
            if harness
                .peer
                .state
                .data
                .lock()
                .requests
                .iter()
                .skip(after)
                .any(|row| row["type"] == kind)
            {
                break;
            }
            tokio::time::sleep(Duration::from_millis(5)).await;
        }
    })
    .await
    .expect("ordinary query did not reach the expected actual HTTP source");
}

fn override_response(harness: &Harness, kind: &str, delay: u64, raw: String) {
    harness
        .peer
        .state
        .data
        .lock()
        .execution
        .as_mut()
        .unwrap()
        .startup_responses
        .insert(kind.into(), (delay, raw));
}

fn requests_since(harness: &Harness, after: usize) -> Vec<Value> {
    harness
        .peer
        .state
        .data
        .lock()
        .requests
        .iter()
        .skip(after)
        .cloned()
        .collect()
}

fn count(requests: &[Value], kind: &str) -> usize {
    requests.iter().filter(|row| row["type"] == kind).count()
}

async fn expire_original_prerequisite(harness: &Harness) {
    let before = harness.scope();
    let original = before["warm_prerequisite_source"].clone();
    assert!(
        original.is_object(),
        "normal original source is missing: {before}"
    );
    let expires = original["effective_expires_ms"].as_u64().unwrap();
    tokio::time::timeout(Duration::from_secs(3), async {
        while now_ms() <= expires {
            tokio::time::sleep(Duration::from_millis(5)).await;
        }
    })
    .await
    .expect("original two-second prerequisite did not expire");

    // A real new financial frame restores current private evidence, not the old source age
    let sequence = before["startup_private_ingress"]["received_sequence"]
        .as_u64()
        .unwrap();
    let published = now_ms();
    harness.peer.state.data.lock().io["time"] = json!(published);
    let frame = harness.peer.state.clearinghouse_frame();
    let raw = frame.to_string();
    harness
        .peer
        .state
        .instructions
        .send(PeerInstruction::Frame(frame))
        .unwrap();
    tokio::time::timeout(Duration::from_secs(2), async {
        loop {
            let scope = harness.scope();
            let delivered = harness
                .peer
                .state
                .data
                .lock()
                .ws_observations
                .iter()
                .any(|row| row["raw_body"] == raw && row["delivered"] == true);
            if delivered
                && scope["startup_private_ingress"]["received_sequence"]
                    .as_u64()
                    .unwrap()
                    > sequence
                && scope["startup_private_ingress"]["is_applied"] == true
                && scope["account"]["ws_source_time_ms"] == published
            {
                assert_eq!(
                    scope["startup_private_ingress"]["generation"],
                    before["startup_private_ingress"]["generation"]
                );
                assert_eq!(
                    scope["startup_private_ingress"]["epoch"],
                    before["startup_private_ingress"]["epoch"]
                );
                assert_eq!(scope["warm_prerequisite_source"], original);
                assert_eq!(scope["metadata"], before["metadata"]);
                assert!(now_ms() > expires);
                break;
            }
            tokio::time::sleep(Duration::from_millis(5)).await;
        }
    })
    .await
    .expect(
        "actual fresh financial frame did not become fully applied after original source expiry",
    );
}

#[derive(Debug)]
struct CapturedWarm {
    origin: Value,
    policy: Value,
    prerequisite: Value,
    started_ms: u64,
    token: Value,
}

fn issue_captured(harness: &Harness) -> CapturedWarm {
    let before = harness.scope();
    let started_ms = now_ms();
    issue(harness);
    let pending = harness.scope();
    CapturedWarm {
        origin: before["warm_metadata_origin"].clone(),
        policy: before["policy"].clone(),
        prerequisite: if pending["last_query_reused_prerequisite"] == true {
            before["warm_prerequisite_source"].clone()
        } else {
            Value::Null
        },
        started_ms,
        token: harness.scope()["query_token"].clone(),
    }
}

fn assert_original_debt(harness: &Harness, captured: &CapturedWarm) -> Value {
    let scope = harness.scope();
    let debt = &scope["recovery_source_debt"];
    assert_eq!(
        debt["kind"], "unknown_sources",
        "captured raw financial sources were discarded: {scope}"
    );
    let identity = &debt["identity"];
    assert_eq!(identity["generation"], captured.origin["generation"]);
    assert_eq!(identity["epoch"], captured.origin["epoch"]);
    assert_eq!(identity["account_id"], "HYPERLIQUID-ENTROPY");
    assert_eq!(identity["address"], USER);
    assert_eq!(identity["policy"], captured.policy);
    assert!(identity["received_sequence"].as_u64().is_some());
    assert!(identity["query_token"].as_u64().is_some());
    assert!(
        captured.token.as_u64().is_some(),
        "normal pending query did not expose its actual token"
    );
    assert_eq!(identity["query_token"], captured.token);
    let sources = debt["original_sources"].as_array().unwrap();
    assert!(
        !sources.is_empty(),
        "received original bodies were replaced with missing-raw claims"
    );
    let now = now_ms();
    let data = harness.peer.state.data.lock();
    if captured.prerequisite.is_object() {
        let original = &debt["reused_prerequisite_source"];
        assert_eq!(original, &captured.prerequisite);
        assert_eq!(original["account_id"], identity["account_id"]);
        assert_eq!(original["address"], identity["address"]);
        assert_eq!(original["policy"], captured.policy);
        assert_eq!(original["generation"], identity["generation"]);
        assert_eq!(original["epoch"], identity["epoch"]);
        let old_sources = original["sources"]["original_sources"].as_array().unwrap();
        let start = usize::try_from(original["account_source_start"].as_u64().unwrap()).unwrap();
        assert!(old_sources.len() >= start + 8);
        for (index, source) in old_sources.iter().enumerate() {
            let request: Value =
                serde_json::from_str(source["request_json"].as_str().unwrap()).unwrap();
            let received = source["received_ms"].as_u64().unwrap();
            assert!(received <= captured.started_ms && received <= now);
            if (start..start + 8).contains(&index) {
                assert!(received >= original["verification_started_ms"].as_u64().unwrap());
                assert!(received <= original["verification_finished_ms"].as_u64().unwrap());
            }
            assert!(
                data.http_observations.iter().any(|actual| {
                    actual["request"] == request
                        && actual["raw_body"] == source["raw_text"]
                        && actual["started_ms"].as_u64().unwrap() <= received
                }),
                "original prerequisite raw has no actual prior response: {source}"
            );
        }
    }
    for source in sources {
        let request: Value =
            serde_json::from_str(source["request_json"].as_str().unwrap()).unwrap();
        let received = source["received_ms"].as_u64().unwrap();
        assert!(received >= captured.started_ms && received <= now);
        assert!(
            data.http_observations
                .iter()
                .any(|actual| actual["request"] == request
                    && actual["raw_body"] == source["raw_text"]
                    && actual["started_ms"].as_u64().unwrap() <= received),
            "retained source does not match the actual normal peer response: {source}"
        );
    }
    assert_eq!(scope["recovery_complete"], false);
    assert_eq!(scope["native_projection_recovery_required"], true);
    debt.clone()
}

fn cost(requests: &[Value]) -> u32 {
    requests
        .iter()
        .map(|row| match row["type"].as_str().unwrap() {
            "userRole" => 60,
            "clearinghouseState" | "orderStatus" => 2,
            _ => 20,
        })
        .sum()
}

fn assert_current(harness: &Harness, scope: &Value) {
    assert_eq!(scope["recovery_complete"], true, "{scope}");
    assert_eq!(scope["account"]["trusted"], true, "{scope}");
    assert_eq!(scope["native_projection_recovery_required"], false);
    assert_eq!(scope["account"]["private_stream_epoch"], 0);
    assert!(now_ms() - scope["account"]["http_received_time_ms"].as_u64().unwrap() < 2000);
    assert!(
        now_ms()
            - scope["account"]["http_verification_started_time_ms"]
                .as_u64()
                .unwrap()
            < 2000
    );
    assert_eq!(harness.peer.state.active.load(Ordering::SeqCst), 1);
}

async fn actual_post_fill(harness: &mut Harness) -> OrderAny {
    let order = harness.order(
        "E-WARM-ACTUAL",
        OrderSide::Buy,
        "0.12",
        "100",
        false,
        TimeInForce::Ioc,
    );
    harness.submit(&order).unwrap();
    harness.wait_posts(1).await;
    // Wait for the ordinary aggregate ACK recovery to finish, before the peer's actual fill
    tokio::time::timeout(Duration::from_secs(2), async {
        loop {
            if harness.scope()["diagnostic"]
                .as_str()
                .unwrap_or("")
                .contains("RecoveryIncomplete")
            {
                break;
            }
            tokio::time::sleep(Duration::from_millis(5)).await;
        }
    })
    .await
    .expect("aggregate ACK without actual fills must remain incomplete");
    let fill = harness.fill(0, 901, "0.12", "100", "0.001", "0");
    harness.terminal(0, "filled");
    harness.set_position("0.12");
    harness.send_fill(fill);
    harness
        .wait_status(order.client_order_id(), OrderStatus::Filled)
        .await;
    harness.wait_latest_position_snapshot().await;
    assert_eq!(harness.fill_events, 1);
    assert_eq!(
        harness.portfolio.net_position(&order.instrument_id()),
        Decimal::from_str_exact("0.12").unwrap()
    );
    assert_eq!(
        harness
            .cache
            .borrow()
            .order(&order.client_order_id())
            .unwrap()
            .commissions()[&Currency::USDC()]
            .as_decimal(),
        Decimal::from_str_exact("0.001").unwrap()
    );
    order
}

#[tokio::test]
async fn actual_post_fill_warm_query_absorbs_reservation_with_one_final_account_group() {
    let mut harness = normal().await;
    let order = actual_post_fill(&mut harness).await;
    let before = harness.scope();
    let after = harness.peer.state.data.lock().requests.len();
    let started = now_ms();
    issue(&harness);
    assert_eq!(harness.scope()["recovery_complete"], false);
    let result = finished(&harness).await;
    harness.apply_events();
    assert_current(&harness, &result);
    assert_eq!(result["last_query_warm"], true);
    assert_eq!(result["query_completed"], 1);
    let requests = requests_since(&harness, after);
    eprintln!(
        "warm_post_fill_actual_requests={} minimum_weight={}",
        json!(requests),
        cost(&requests)
    );
    eprintln!(
        "warm_post_fill_actual_http_sources={}",
        json!(harness.peer.state.data.lock().http_observations)
    );
    eprintln!(
        "warm_post_fill_actual_ws_sources={}",
        json!(harness.peer.state.data.lock().ws_observations)
    );
    assert_eq!(count(&requests, "frontendOpenOrders"), 1);
    assert_eq!(count(&requests, "userFills"), 1);
    assert_eq!(count(&requests, "orderStatus"), 1);
    assert_eq!(count(&requests, "userRole"), 0);
    assert_eq!(count(&requests, "userAbstraction"), 0);
    assert_eq!(count(&requests, "userDexAbstraction"), 0);
    assert_eq!(count(&requests, "meta"), 0);
    assert_eq!(count(&requests, "spotMeta"), 0);
    assert_eq!(count(&requests, "clearinghouseState"), 1);
    assert_eq!(count(&requests, "perpDexs"), 0);
    assert_eq!(count(&requests, "activeAssetData"), 0);
    assert_eq!(cost(&requests), 44);
    assert_eq!(result["last_query_reused_prerequisite"], true);
    assert!(before["warm_prerequisite_source"].is_object());
    assert_eq!(
        result["warm_prerequisite_source"],
        before["warm_prerequisite_source"]
    );
    assert_eq!(
        result["account"]["http_verification_started_time_ms"],
        before["warm_prerequisite_source"]["verification_started_ms"]
    );
    assert!(result["account"]["http_received_time_ms"].as_u64().unwrap() >= started);
    assert_eq!(
        result["owned_intents"][order.client_order_id().as_str()]["phase"],
        "terminal"
    );
    assert_eq!(
        result["owned_intents"][order.client_order_id().as_str()]["reservation"],
        "0"
    );
    assert_eq!(result["actual_fills"], before["actual_fills"]);
    assert_eq!(harness.fill_events, 1);
    assert_eq!(
        harness
            .cache
            .borrow()
            .order(&order.client_order_id())
            .unwrap()
            .trade_ids()
            .len(),
        1
    );
    assert_eq!(
        harness.portfolio.net_position(&order.instrument_id()),
        Decimal::from_str_exact("0.12").unwrap()
    );
    assert_eq!(
        harness
            .cache
            .borrow()
            .order(&order.client_order_id())
            .unwrap()
            .commissions()[&Currency::USDC()]
            .as_decimal(),
        Decimal::from_str_exact("0.001").unwrap()
    );
    harness.stop().await;
}

#[tokio::test]
async fn equivalent_normal_queries_singleflight_without_extra_reads_or_borrowed_ready() {
    let mut harness = normal().await;
    let before_metadata = harness.scope()["metadata"].clone();
    override_response(&harness, "frontendOpenOrders", 120, "[]".into());
    let after = harness.peer.state.data.lock().requests.len();
    for _ in 0..4 {
        issue(&harness);
    }
    assert_eq!(harness.scope()["recovery_complete"], false);
    assert_eq!(harness.scope()["account"]["trusted"], false);
    let result = finished(&harness).await;
    harness.apply_events();
    assert_current(&harness, &result);
    assert_eq!(result["last_query_warm"], true);
    assert_eq!(
        result["query_completed"], 1,
        "equivalent queries did not share one flight"
    );
    let requests = requests_since(&harness, after);
    assert_eq!(count(&requests, "frontendOpenOrders"), 1);
    assert_eq!(count(&requests, "userRole"), 0);
    assert_eq!(count(&requests, "activeAssetData"), 2);
    assert_eq!(cost(&requests), 82);
    assert_eq!(result["last_query_reused_prerequisite"], true);
    assert_eq!(
        result["metadata"], before_metadata,
        "warm leverage counterproof relabeled original metadata"
    );
    assert_eq!(harness.fill_events, 0);
    assert_eq!(harness.peer.state.writes.load(Ordering::SeqCst), 0);
    harness.stop().await;
}

#[rstest]
#[case("out_of_policy")]
#[case("user")]
#[case("coin")]
#[case("cross")]
#[case("changed_in_policy")]
#[case("conflicting_repeat")]
#[tokio::test]
async fn flat_warm_requires_actual_repeated_selected_leverage_counterproof(#[case] fault: &str) {
    let mut harness = Harness::with_configuration(
        "100",
        |peer| {
            peer.record_http = true;
        },
        |config, directory| {
            configure(config, directory);
            if matches!(fault, "changed_in_policy" | "conflicting_repeat") {
                let mut selected: Value =
                    serde_json::from_str(config.io_execution_policy_json.as_ref().unwrap())
                        .unwrap();
                selected["max_leverage"] = json!(2);
                config.io_execution_policy_json = Some(selected.to_string());
            }
        },
    )
    .await;
    let before = harness.scope();
    let mut actual = harness
        .peer
        .state
        .data
        .lock()
        .execution
        .as_ref()
        .unwrap()
        .asset_data
        .clone();
    let original = actual.clone();
    match fault {
        "out_of_policy" | "changed_in_policy" | "conflicting_repeat" => {
            actual["leverage"]["value"] = json!(2);
        }
        "user" => actual["user"] = json!("0x0000000000000000000000000000000000000001"),
        "coin" => actual["coin"] = json!("io:OTHER"),
        "cross" => actual["leverage"]["type"] = json!("cross"),
        _ => unreachable!(),
    }
    {
        let mut data = harness.peer.state.data.lock();
        let execution = data.execution.as_mut().unwrap();
        if fault == "conflicting_repeat" {
            execution.warm_asset_data_pages =
                VecDeque::from([original.to_string(), actual.to_string()]);
        } else {
            execution.asset_data = actual;
        }
    }
    let after = harness.peer.state.data.lock().requests.len();
    let captured = issue_captured(&harness);
    let result = finished(&harness).await;
    harness.apply_events();
    assert_eq!(
        result["recovery_complete"], false,
        "flat account omitted selected leverage evidence: {fault}: {result}"
    );
    assert_original_debt(&harness, &captured);
    let requests = requests_since(&harness, after);
    assert!((1..=2).contains(&count(&requests, "activeAssetData")));
    if fault == "conflicting_repeat" {
        assert_eq!(count(&requests, "activeAssetData"), 2);
    }
    assert_eq!(
        result["metadata"], before["metadata"],
        "failed actual leverage source refreshed metadata time/capacity/mark"
    );
    assert_eq!(result["actual_fills"], before["actual_fills"]);
    assert_eq!(result["owned_intents"], before["owned_intents"]);
    assert_eq!(harness.fill_events, 0);
    assert_eq!(
        harness
            .portfolio
            .net_position(&InstrumentId::from(SELECTED)),
        Decimal::ZERO
    );
    assert_eq!(harness.peer.state.writes.load(Ordering::SeqCst), 0);
    harness.stop().await;
}

#[tokio::test]
async fn flat_warm_leverage_counterproof_never_refreshes_original_metadata_capacity_or_mark() {
    let mut harness = normal().await;
    let before = harness.scope();
    {
        let mut data = harness.peer.state.data.lock();
        let actual = &mut data.execution.as_mut().unwrap().asset_data;
        actual["maxTradeSzs"] = json!(["2", "2"]);
        actual["availableToTrade"] = json!(["200", "200"]);
        actual["markPx"] = json!("101");
    }
    let after = harness.peer.state.data.lock().requests.len();
    issue(&harness);
    let result = finished(&harness).await;
    harness.apply_events();
    assert_current(&harness, &result);
    let requests = requests_since(&harness, after);
    assert_eq!(count(&requests, "activeAssetData"), 2);
    assert_eq!(cost(&requests), 82);
    assert_eq!(result["last_query_reused_prerequisite"], true);
    assert_eq!(
        result["metadata"], before["metadata"],
        "leverage counterproof widened or relabeled original metadata"
    );
    assert_eq!(result["actual_fills"], before["actual_fills"]);

    let after = economics(&harness);
    let previous_end = before["coverage"]
        .as_array()
        .unwrap()
        .iter()
        .filter_map(|coverage| coverage["end_ms"].as_u64())
        .max()
        .unwrap();
    let new_coverage = after["coverage"]
        .as_array()
        .unwrap()
        .iter()
        .filter(|coverage| {
            coverage["end_ms"]
                .as_u64()
                .is_some_and(|end| end > previous_end)
        })
        .collect::<Vec<_>>();
    assert_eq!(
        new_coverage.len(),
        2,
        "hard history demand did not record both actual endpoint outcomes: {after}"
    );
    let history_requests =
        count(&requests, "userFunding") + count(&requests, "userNonFundingLedgerUpdates");
    let deadline = "Economic history total deadline exhausted (quota/retries/metadata included)";
    for coverage in &new_coverage {
        assert_eq!(coverage["complete"], false);
        assert_eq!(coverage["generation"], before["coverage"][0]["generation"]);
        assert_eq!(coverage["epoch"], before["coverage"][0]["epoch"]);
        assert_eq!(coverage["end_ms"], new_coverage[0]["end_ms"]);
        if history_requests == 0 {
            assert_eq!(
                coverage["diagnostic"], deadline,
                "quota refusal lost the exact native deadline: {coverage}"
            );
            assert_eq!(coverage["pages"], 0);
            assert_eq!(coverage["records"], 0);
        } else {
            let diagnostic = coverage["diagnostic"].as_str().unwrap().to_lowercase();
            assert!(
                diagnostic.contains("deadline")
                    || (diagnostic.contains("incomplete")
                        && (diagnostic.contains("timed out") || diagnostic.contains("timeout"))),
                "actual history transport failure is not a genuine bounded timeout: {coverage}"
            );
        }
    }
    assert!(
        new_coverage
            .iter()
            .any(|coverage| coverage["endpoint"] == "userFunding")
    );
    assert!(
        new_coverage
            .iter()
            .any(|coverage| coverage["endpoint"] == "userNonFundingLedgerUpdates")
    );
    for previous in before["coverage"].as_array().unwrap() {
        assert!(
            after["coverage"].as_array().unwrap().contains(previous),
            "prior finite history coverage was relabeled: {previous}"
        );
    }
    if history_requests == 0 {
        assert!(count(&requests, "userRole") >= 2 && count(&requests, "clearinghouseState") >= 2);
        assert!(count(&requests, "frontendOpenOrders") >= 1 && count(&requests, "userFills") >= 1);
        assert!(count(&requests, "perpDexs") >= 1 && count(&requests, "activeAssetData") >= 2);
        assert!(
            count(&requests, "meta") >= 4 && count(&requests, "spotMeta") >= 4,
            "no history transport arrived without actual cold and economic metadata reads: {}",
            json!(requests)
        );
        let data = harness.peer.state.data.lock();
        for endpoint in ["meta", "spotMeta"] {
            assert!(
                data.http_observations
                    .iter()
                    .any(|response| response["request"]["type"] == endpoint
                        && response["status"] == 200
                        && response["started_ms"].as_u64().is_some_and(
                            |start| start >= new_coverage[0]["end_ms"].as_u64().unwrap()
                        )),
                "missing actual completed economic metadata read after the native demand end: {endpoint}"
            );
        }
    }
    eprintln!(
        "hard_history_bounded_outcome={}",
        json!({"history_requests":history_requests,
        "actual_request_count":requests.len(),"new_coverage":new_coverage,
        "proof_ready":harness.scope()["recovery_complete"],"signed_writes":harness.peer.state.writes.load(Ordering::SeqCst)})
    );
    assert_eq!(
        after["report"]["receipts"], before["report"]["receipts"],
        "Unknown history demand reapplied financial receipts"
    );
    assert!(!after["raw_envelopes"].as_object().unwrap().is_empty());
    for (key, raw) in before["raw_envelopes"].as_object().unwrap() {
        assert_eq!(&after["raw_envelopes"][key], raw);
    }
    assert_eq!(harness.fill_events, 0);
    assert_eq!(harness.peer.state.writes.load(Ordering::SeqCst), 0);
    assert_eq!(harness.scope()["actions"], 0);
    harness.stop().await;
}

#[tokio::test]
async fn same_zero_epoch_new_factory_reader_cannot_apply_retired_readers_queued_source() {
    let mut retired = normal().await;
    let epoch = retired.scope()["account"]["private_stream_epoch"].clone();
    assert_eq!(epoch, 0);
    let captured = issue_captured(&retired);
    staged_owner_callback(&retired).await;
    retired.client.disconnect().await.unwrap();
    while retired.receiver.try_recv().is_ok() {}
    // A genuinely new normal factory starts its own reader at the same numeric epoch
    let mut replacement = normal().await;
    assert_eq!(
        replacement.scope()["account"]["private_stream_epoch"],
        epoch
    );
    let current = replacement.scope();
    retired.apply_time_events();
    tokio::time::sleep(Duration::from_millis(20)).await;
    assert_eq!(retired.scope()["recovery_complete"], false);
    let retained = assert_original_debt(&retired, &captured);
    assert!(
        retired.receiver.try_recv().is_err(),
        "retired source emitted financial state on a replacement reader"
    );
    assert_eq!(replacement.scope()["actual_fills"], current["actual_fills"]);
    assert_eq!(replacement.fill_events, 0);
    assert_eq!(retired.fill_events, 0);
    assert_eq!(retired.scope()["recovery_source_debt"], retained);
    assert!(
        replacement.scope()["recovery_source_debt"].is_null(),
        "retired raw evidence was adopted by a new reader"
    );
    replacement.stop().await;
    retired.stop().await;
}
