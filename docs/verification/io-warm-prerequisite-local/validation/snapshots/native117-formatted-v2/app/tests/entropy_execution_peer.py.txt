class ExecutionPeer:
    """Real servers; one selected instrument, fixed finite request/action/frame caps."""

    def __init__(self, venue, *, fault="none"):
        if venue not in {"HYPERLIQUID", "ASTER"}:
            raise ValueError("unsupported synthetic venue")
        self.venue, self.fault = venue, fault
        self.requests, self.commands, self.frames, self.errors = [], [], [], []
        self.orders, self.fills, self.sessions, self.workers = {}, [], set(), set()
        self.position = Decimal(0)
        self.sequence, self.actions, self.connections = 100, 0, 0
        self.trade_sequence = 60000
        self.executions = []
        self.stopping = False
        self.tasks = set()

    @staticmethod
    def stamp():
        return time.time_ns() // 1_000_000

    async def __aenter__(self):
        from websockets.asyncio.server import serve

        self.http_server = await asyncio.start_server(
            self.request, "127.0.0.1", 0, limit=16384
        )
        self.ws_server = await serve(
            self.stream, "127.0.0.1", 0, max_size=65536, max_queue=16
        )
        self.http_url = (
            f"http://127.0.0.1:{self.http_server.sockets[0].getsockname()[1]}"
        )
        self.ws_url = f"ws://127.0.0.1:{self.ws_server.sockets[0].getsockname()[1]}"
        return self

    async def __aexit__(self, *_):
        self.stopping = True
        for task in self.tasks:
            task.cancel()
        await asyncio.gather(*self.tasks, return_exceptions=True)
        self.ws_server.close()
        self.http_server.close()
        await asyncio.wait_for(self.ws_server.wait_closed(), 5)
        await self.http_server.wait_closed()
        if self.workers:
            await asyncio.wait_for(
                asyncio.gather(*self.workers, return_exceptions=True), 5
            )

    def aster_order(self, method, params):
        first = lambda name, default=None: params.get(name, [default])[0]
        if first("symbol") != "SNDKUSD1":
            raise ValueError("unexpected order symbol")
        if method in {"GET", "DELETE"}:
            row = next(
                (
                    row
                    for row in self.orders.values()
                    if row["clientOrderId"] == first("origClientOrderId")
                    or str(row["orderId"]) == first("orderId")
                ),
                None,
            )
            if row is None:
                return {"code": -2013, "msg": "Order does not exist."}
            if method == "DELETE":
                self.actions += 1
                row.update(status="CANCELED", updateTime=self.stamp())
            return row
        if method != "POST" or self.actions >= 6:
            raise ValueError("unsupported/budgeted order action")
        self.actions += 1
        if self.fault == "reject" or (
            self.fault == "cleanup_reject" and first("reduceOnly") == "true"
        ):
            return {"code": -2010, "msg": "synthetic bounded rejection"}
        identifier = first("newClientOrderId")
        if not identifier or identifier in self.orders:
            raise ValueError("unowned or repeated client id")
        row = {
            "symbol": "SNDKUSD1",
            "orderId": 80000 + self.actions,
            "clientOrderId": identifier,
            "origQty": first("quantity"),
            "executedQty": "0",
            "price": first("price"),
            "avgPrice": "0",
            "status": "NEW",
            "timeInForce": first("timeInForce"),
            "type": "LIMIT",
            "side": first("side"),
            "positionSide": "BOTH",
            "reduceOnly": first("reduceOnly") == "true",
            "time": self.stamp(),
            "updateTime": self.stamp(),
        }
        self.orders[identifier] = row
        self.spawn(self.aster_fill(row))
        return row

    def spawn(self, awaitable):
        task = asyncio.create_task(awaitable)
        self.tasks.add(task)
        task.add_done_callback(self.tasks.discard)

    async def request(self, reader, writer):
        task = asyncio.current_task()
        self.workers.add(task)
        request_record = None
        try:
            header = await asyncio.wait_for(reader.readuntil(b"\r\n\r\n"), 2)
            lines = header.decode("ascii").split("\r\n")
            method, target, _ = lines[0].split(" ")
            headers = {
                key.lower(): value.strip()
                for key, value in (
                    line.split(":", 1) for line in lines[1:] if ":" in line
                )
            }
            length = int(headers.get("content-length", "0"))
            if (
                len(header) > 16384
                or not 0 <= length <= 65536
                or len(self.requests) >= 150
            ):
                raise ValueError("peer request capacity exceeded")
            if not headers.get("host", "").startswith("127.0.0.1:"):
                raise ValueError("nonloopback host")
            payload = (
                await asyncio.wait_for(reader.readexactly(length), 2) if length else b""
            )
            parsed = urlsplit(target)
            body = (
                json.loads(payload) if self.venue == "HYPERLIQUID" and payload else None
            )
            # Synthetic signatures remain local; evidence deliberately strips private headers.
            params = parse_qs(parsed.query)
            if self.venue == "ASTER" and payload:
                params.update(parse_qs(payload.decode()))
            recorded_params = {
                key: value
                for key, value in params.items()
                if key not in {"signature", "user", "signer"}
            }
            request_record = {
                "method": method,
                "path": parsed.path,
                "body": body,
                "params": recorded_params,
            }
            self.requests.append(request_record)
            if self.venue == "HYPERLIQUID":
                if method != "POST" or parsed.path != "/info":
                    raise ValueError("only native WS action transport is permitted")
                result = self.hl_info(body)
            else:
                if not parsed.path.startswith("/fapi/"):
                    raise ValueError("unexpected Aster path")
                result = self.aster_response(method, parsed.path, params)
                if (
                    self.fault == "missing_response"
                    and parsed.path.endswith("/order")
                    and method == "POST"
                ):
                    await asyncio.sleep(3)
                    writer.close()
                    return
            status = (
                "400 Bad Request"
                if isinstance(result, dict) and result.get("code", 0) < 0
                else "200 OK"
            )
        except (
            ValueError,
            RuntimeError,
            TypeError,
            KeyError,
            AttributeError,
            OSError,
            ArithmeticError,
        ) as exc:
            self.errors.append(f"{type(exc).__name__}: {exc}")
            status, result = "400 Bad Request", {"error": type(exc).__name__}
        try:
            raw = json.dumps(result, separators=(",", ":")).encode()
            if len(raw) > 65536:
                raise ValueError("peer response body cap exceeded")
            if request_record is not None:
                request_record["response"] = {
                    "status": status,
                    "body": json.loads(raw),
                }
            writer.write(
                f"HTTP/1.1 {status}\r\nContent-Type: application/json\r\nContent-Length: {len(raw)}\r\nConnection: close\r\n\r\n".encode()
                + raw
            )
            await writer.drain()
        finally:
            writer.close()
            await writer.wait_closed()
            self.workers.discard(task)

    async def broadcast(self, frame):
        if len(self.frames) >= 1500:
            raise ValueError("peer frame cap exceeded")
        self.frames.append(json.loads(json.dumps(frame)))
        await asyncio.gather(
            *(
                ws.send(json.dumps(frame))
                for ws, private in tuple(self.sessions)
                if private
            ),
            return_exceptions=True,
        )

    async def send_frame(self, ws, frame):
        if len(self.frames) >= 1500:
            raise ValueError("peer frame cap exceeded")
        self.frames.append(json.loads(json.dumps(frame)))
        await ws.send(json.dumps(frame))

    async def hl_action(self, ws, command):
        action = command["request"]["payload"]["action"]
        if self.actions >= 6:
            raise ValueError("peer action budget exceeded")
        self.actions += 1
        if action["type"] == "order":
            if len(action["orders"]) != 1:
                raise ValueError("wrong native order count")
            wire = action["orders"][0]
            if wire["a"] != 110000 or wire["c"] in self.orders:
                raise ValueError("wrong native asset/ownership/retransmission")
            # This fault represents a received but unexecuted request with no result.
            # The native client must still retain its own worst-case ambiguity.
            if self.fault == "missing_response":
                return
            rejected = self.fault == "reject" or (
                self.fault == "cleanup_reject" and wire["r"]
            )
            if rejected:
                statuses = [{"error": "synthetic bounded rejection"}]
            else:
                row = self.hl_execute(wire)
                oid = row["oid"]
                statuses = [
                    {"filled": {"oid": oid, "totalSz": wire["s"], "avgPx": wire["p"]}}
                ]
                self.spawn(self.hl_fill(row))
        elif action["type"] in {"cancel", "cancelByCloid"}:
            if self.fault == "missing_response":
                return
            statuses = (
                [{"error": "synthetic bounded rejection"}]
                if self.fault == "reject"
                else ["success"]
            )
        else:
            raise ValueError("unsupported action")
        await self.send_frame(
            ws,
            {
                "channel": "post",
                "data": {
                    "id": command["id"],
                    "response": {
                        "type": "action",
                        "payload": {
                            "status": "ok",
                            "response": {
                                "type": "order"
                                if action["type"] == "order"
                                else "cancel",
                                "data": {"statuses": statuses},
                            },
                        },
                    },
                },
            },
        )

    def hl_execute(self, wire):
        """Execute once before Filled ACK; delayed publication is a separate event."""
        quantity = Decimal(wire["s"])
        price = Decimal(wire["p"])
        change = quantity if wire["b"] else -quantity
        start = self.position
        if (
            not quantity.is_finite()
            or not price.is_finite()
            or quantity <= 0
            or price <= 0
        ):
            raise ValueError("invalid synthetic quantity/price")
        if wire["r"] and (not start or start * change >= 0 or abs(change) > abs(start)):
            raise ValueError("reduce-only request does not reduce actual position")
        oid, cloid = 70000 + self.actions, wire["c"]
        executed_ms = self.stamp()
        self.trade_sequence += 1
        tid = self.trade_sequence
        self.position += change
        self.orders[cloid] = {
            "order": {
                "coin": "io:SNDK",
                "side": "B" if wire["b"] else "A",
                "limitPx": wire["p"],
                "sz": "0",
                "origSz": wire["s"],
                "oid": oid,
                "timestamp": executed_ms,
                "cloid": cloid,
                "reduceOnly": wire["r"],
                "orderType": "Limit",
                "tif": wire["t"]["limit"]["tif"],
            },
            "status": "filled",
            "statusTimestamp": executed_ms,
        }
        row = {
            "coin": "io:SNDK",
            "px": wire["p"],
            "sz": str(quantity),
            "side": "B" if wire["b"] else "A",
            "time": executed_ms,
            "startPosition": str(start),
            "dir": ("Close Long" if start > 0 else "Close Short")
            if wire["r"]
            else ("Open Long" if wire["b"] else "Open Short"),
            "closedPnl": "0",
            "hash": f"0x{tid:064x}",
            "oid": oid,
            "crossed": True,
            "fee": "2" if self.fault == "loss_fee" else "0.001",
            "builderFee": "0.0002",
            "tid": tid,
            "feeToken": "USDC",
            "cloid": wire["c"],
        }
        self.executions.append(
            {"fill": json.loads(json.dumps(row)), "published_ms": None}
        )
        return row

    async def hl_fill(self, executed_row):
        await asyncio.sleep(0.15)
        # No new execution, time/identity allocation, or position mutation here.
        row = json.loads(json.dumps(executed_row))
        self.fills.append(row)
        next(item for item in self.executions if item["fill"]["tid"] == row["tid"])[
            "published_ms"
        ] = self.stamp()
        await self.broadcast({"channel": "user", "data": {"fills": [row]}})
        await self.broadcast(
            {
                "channel": "clearinghouseState",
                "data": {
                    "user": USER,
                    "dex": "io",
                    "clearinghouseState": self.clearinghouse(),
                },
            }
        )

    async def stream(self, ws):
        self.connections += 1
        path = ws.request.path
        private = self.venue == "HYPERLIQUID" or "entropy43-owned-synthetic" in path
        self.sessions.add((ws, private))
        subscriptions = set()
        producer = asyncio.create_task(self.produce(ws, subscriptions, private))
        try:
            async for raw in ws:
                if len(self.commands) >= 100 or len(raw) > 65536:
                    raise ValueError("peer command cap exceeded")
                command = json.loads(raw)
                self.commands.append(command)
                if self.venue == "HYPERLIQUID":
                    method = command["method"]
                    if method == "post":
                        await self.hl_action(ws, command)
                    elif method == "ping":
                        await ws.send(json.dumps({"channel": "pong"}))
                    elif method in {"subscribe", "unsubscribe"}:
                        sub = command["subscription"]
                        if method == "subscribe":
                            subscriptions.add(sub["type"])
                        else:
                            subscriptions.discard(sub["type"])
                        await ws.send(
                            json.dumps(
                                {"channel": "subscriptionResponse", "data": command}
                            )
                        )
                    else:
                        raise ValueError("unexpected HL command")
                else:
                    if command["method"] not in {"SUBSCRIBE", "UNSUBSCRIBE"}:
                        raise ValueError("unexpected Aster public command")
                    subscriptions.update(command["params"])
                    await ws.send(json.dumps({"result": None, "id": command["id"]}))
        except (
            ValueError,
            RuntimeError,
            TypeError,
            KeyError,
            AttributeError,
            OSError,
            ArithmeticError,
            ConnectionClosed,
        ) as exc:
            if not isinstance(exc, ConnectionClosed):
                self.errors.append(f"{type(exc).__name__}: {exc}")
        finally:
            producer.cancel()
            await asyncio.gather(producer, return_exceptions=True)
            self.sessions.discard((ws, private))

    async def produce(self, ws, subscriptions, private):
        last_private_ms = 0
        while True:
            await asyncio.sleep(0.1)
            if self.venue == "HYPERLIQUID":
                if "l2Book" in subscriptions:
                    await self.send_frame(
                        ws, {"channel": "l2Book", "data": self.hl_book()}
                    )
                if (
                    "clearinghouseState" in subscriptions
                    and self.stamp() - last_private_ms >= 1000
                ):
                    await self.send_frame(
                        ws,
                        {
                            "channel": "clearinghouseState",
                            "data": {
                                "user": USER,
                                "dex": "io",
                                "clearinghouseState": self.clearinghouse(),
                            },
                        },
                    )
                    last_private_ms = self.stamp()
            elif not private and subscriptions:
                self.sequence += 1
                await self.send_frame(
                    ws,

                        "e": "depthUpdate",
                        "E": self.stamp(),
                        "T": self.stamp(),
                        "s": "SNDKUSD1",
                        "U": self.sequence,
