"""Install only the separately validated corrected candidate; keep old identities."""

import hashlib
import json
import subprocess
import sys
from pathlib import Path


ROOT = Path(__file__).resolve().parent
NATIVE = Path("E:/persarb/worktrees/entropy-warm-prerequisites")
APP = Path("E:/persarb/worktrees/entropy-execution-app")
ARTIFACT = Path("E:/persarb/_tmp/entropy117-corrected-wheel-nextest-20261009")


def main():
    provenance_path = ARTIFACT / "native-provenance.json"
    provenance = json.loads(provenance_path.read_text(encoding="utf-8"))
    implementation = json.loads((ROOT / "native117-economic-correction-implementation-commit.json").read_text(encoding="utf-8"))
    build = json.loads((ROOT / "native117-corrected-source-bound-wheel-build-v1.json").read_text(encoding="utf-8"))
    assert build["exit_code"] == 0 and provenance["source_binding"] == "verified"
    assert provenance["declared_native"]["commit"] == implementation["commit"]
    assert provenance["declared_native"]["tree"] == implementation["tree"]
    assert not provenance["declared_native"]["dirty_count"]
    actual_commit = subprocess.check_output(["git", "rev-parse", "HEAD"], cwd=NATIVE, text=True).strip()
    assert actual_commit == implementation["commit"]
    assert not subprocess.check_output(["git", "status", "--porcelain"], cwd=NATIVE).strip()
    wheel = ARTIFACT / provenance["wheel"]["filename"]
    assert hashlib.sha256(wheel.read_bytes()).hexdigest() == provenance["wheel"]["sha256"]
    for name, directory in (("native", NATIVE), ("app", APP)):
        interpreter = directory / ".venv" / "Scripts" / "python.exe"
        output = ROOT / f"native117-corrected-{name}-installed-identity-v1.json"
        assert not output.exists()
        command = [str(interpreter), "-X", "utf8", str(APP / "scripts" / "install_native.py"),
                   "--python", str(interpreter), "--local-override", "--native-root", str(NATIVE),
                   "--wheel", str(wheel), "--provenance", str(provenance_path),
                   "--sha256", provenance["wheel"]["sha256"],
                   "--additional-adapter", "hyperliquid", "--additional-adapter", "aster", "--output", str(output)]
        subprocess.run([sys.executable, "-X", "utf8", str(ROOT / "run_capture.py"),
                        f"native117-corrected-{name}-installed-wheel-v1", "--", *command], cwd=APP, check=True)


if __name__ == "__main__":
    main()
