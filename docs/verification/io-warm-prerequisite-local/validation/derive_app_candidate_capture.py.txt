"""Derive the installed-candidate identity capture for the new native117 build."""

import datetime as dt
import hashlib
import json
from pathlib import Path


DIRECTORY = Path(__file__).resolve().parent


def main():
    source = Path("E:/persarb/_tmp/entropy115-validation/capture_app_candidate_v2.py")
    target = DIRECTORY / "capture_app_candidate.py"
    output = DIRECTORY / "native117-app-capture-helper-derivation.json"
    assert not target.exists() and not output.exists()
    before = source.read_bytes()
    text = before.decode("utf-8")
    replacements = (
        ("E:/persarb/worktrees/entropy-startup-account-proof", "E:/persarb/worktrees/entropy-warm-prerequisites"),
        ("E:/persarb/_tmp/entropy115-wheel-nextest-20261009", "E:/persarb/_tmp/entropy117-wheel-nextest-20261009"),
    )
    for old, new in replacements:
        assert text.count(old) == 1, old
        text = text.replace(old, new)
    after = text.encode("utf-8")
    target.write_bytes(after)
    record = {"captured_utc": dt.datetime.now(dt.timezone.utc).isoformat(),
              "source": str(source), "source_sha256": hashlib.sha256(before).hexdigest(),
              "target": str(target), "target_sha256": hashlib.sha256(after).hexdigest(),
              "replacements": replacements, "test_executions": 0, "candidate_captures": 0}
    output.write_text(json.dumps(record, indent=2) + "\n", encoding="utf-8")
    print(json.dumps(record))


if __name__ == "__main__":
    main()
