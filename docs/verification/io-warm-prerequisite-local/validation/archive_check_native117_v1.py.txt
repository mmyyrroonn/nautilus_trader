"""Check every copied archive byte without executing tests or native code."""
import hashlib
import json
from datetime import datetime, timezone
from pathlib import Path

root = Path(__file__).resolve().parent
archive = Path("E:/persarb/worktrees/entropy-warm-prerequisites/docs/verification/io-warm-prerequisite-local")
output = root / "native117-archive-check-final-v1.json"
assert not output.exists()
started = datetime.now(timezone.utc).isoformat()
manifest_path = archive / "manifest.json"
manifest = json.loads(manifest_path.read_text(encoding="utf-8"))
assert manifest["implementation_commit"] == "b07a7e6e6bfc7613aa12c77cdee1110e25790ca1"
assert manifest["preserved_first_implementation"] == "56705ce873d5fc4e568e5f9ea8f2c614d5d35230"
allowed_roots = [root,
    Path("E:/persarb/_tmp/app43-native117-first-paired-20261009"),
    Path("E:/persarb/_tmp/app43-native117-corrected-first-paired-20261009"),
    Path("E:/persarb/_tmp/app43-native117-corrected-remaining-scenarios-20261009"),
    Path("E:/persarb/_tmp/entropy117-wheel-nextest-20261009"),
    Path("E:/persarb/_tmp/entropy117-corrected-wheel-nextest-20261009")]
seen = set()
total_bytes = 0
for row in manifest["files"]:
    relative = Path(row["path"])
    assert not relative.is_absolute() and ".." not in relative.parts
    assert row["path"] not in seen
    seen.add(row["path"])
    target = archive / relative
    assert target.resolve().is_relative_to(archive.resolve()) and not target.is_symlink()
    assert not any(part.startswith(".env") or part in {".venv", ".git", "__pycache__"} for part in relative.parts)
    assert target.suffix not in {".exe", ".dll", ".pyd", ".whl"}
    data = target.read_bytes()
    assert len(data) == row["bytes"] and hashlib.sha256(data).hexdigest() == row["sha256"], row["path"]
    total_bytes += len(data)
    if "source_path" in row:
        source = Path(row["source_path"])
        assert any(source.resolve().is_relative_to(base.resolve()) for base in allowed_roots)
        assert source.stat().st_size == row["bytes"] and hashlib.sha256(source.read_bytes()).hexdigest() == row["sha256"], row["source_path"]
actual_files = {p.relative_to(archive).as_posix() for p in archive.rglob("*") if p.is_file()}
assert actual_files == seen | {"manifest.json"}
regression = json.loads((archive / "regression-summary.json").read_text(encoding="utf-8"))
assert (regression["passed"], regression["failed"], regression["ignored"]) == (1435, 0, 18)
scenarios = json.loads((archive / "application-scenarios.json").read_text(encoding="utf-8"))
assert scenarios["all_seven_scenarios_passed"] is True
assert scenarios["distinct_corrected_parameters"] == 7 and scenarios["actual_corrected_commands"] == 2
assert scenarios["application_issue43_complete"] is False and scenarios["real_venue_e6_verified"] is False
checks = json.loads((archive / "corrected-candidate-checks.json").read_text(encoding="utf-8"))
lint = checks["inherited_clippy_source_proof"]
assert lint["new_diagnostics"] == [] and lint["distinct_inherited_diagnostics"] == 10
assert lint["targeted_overlapping_diagnostics"] == 6 and lint["actual_command_results"] == ["FAIL"] * 3
for name in ("validation/reviewer-native117-corrected-all7-independent-v2.md.txt",
             "validation/reviewer-native117-economic-gap-formatted-v7-review-v1.md.txt",
             "validation/io-warm-prerequisite-contract-draft-v2.md.txt",
             "validation/io-warm-prerequisite-acceptance-draft-v2.md.txt"):
    assert name in seen, name
result = {"started_utc": started, "finished_utc": datetime.now(timezone.utc).isoformat(),
    "archive": archive.as_posix(), "manifest_sha256": hashlib.sha256(manifest_path.read_bytes()).hexdigest(),
    "manifest_records": len(seen), "actual_files": len(actual_files), "copied_bytes": total_bytes,
    "all_recorded_hashes_equal": True, "all_original_copy_sources_unchanged": True,
    "native_passed": 1435, "native_ignored": 18, "original_app_parameters_passed": 7,
    "actual_app_commands": 2, "distinct_inherited_lint_failures": 10,
    "application_issue43_complete": False, "real_venue_e6_verified": False,
    "tests_executed": 0, "native_imports": 0,
    "check_script_sha256": hashlib.sha256(Path(__file__).read_bytes()).hexdigest()}
output.write_text(json.dumps(result, indent=2) + "\n", encoding="utf-8")
print(json.dumps(result))
