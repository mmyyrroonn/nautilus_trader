"""Read actual remote comments before any retry of a failed progress mutation."""

import argparse
import json
from pathlib import Path

from github_capture import api


DIRECTORY = Path(__file__).resolve().parent


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("failed_capture")
    parser.add_argument("label")
    args = parser.parse_args()
    failed = json.loads((DIRECTORY / (args.failed_capture + ".json")).read_text(encoding="utf-8"))
    payload = failed["args"]
    path = f"repos/{payload['repo_full_name']}/issues/{payload['pr_number']}/comments"
    comments = []
    for page in range(1, 21):
        rows = api(f"{path}?per_page=100&page={page}", f"{args.label}-remote-check-page-{page}")
        comments.extend(rows)
        if len(rows) < 100:
            break
    else:
        raise ValueError("Comment pagination bound reached")
    present = [row for row in comments if row["body"].strip() == payload["comment"].strip()]
    assert len(present) <= 1, "Duplicate comment already present; inspect before posting"
    if present:
        result = present[0]
        created = False
    else:
        result = api(path, args.label + "-post-once", "POST", {"body": payload["comment"]})
        created = True
    output = DIRECTORY / (args.label + "-result.json")
    assert not output.exists()
    record = {"failed_capture": args.failed_capture, "checked_comments": len(comments),
              "matches_before_mutation": len(present), "created": created,
              "id": result["id"], "url": result["html_url"], "body": result["body"]}
    output.write_text(json.dumps(record, indent=2) + "\n", encoding="utf-8")
    print(json.dumps({key: value for key, value in record.items() if key != "body"}))


if __name__ == "__main__":
    main()
