    $fullPath = Join-Path $rootPath $entry.Name
    $actual = (Get-FileHash -LiteralPath $fullPath -Algorithm SHA256).Hash.ToLowerInvariant()
    $size = (Get-Item -LiteralPath $fullPath).Length
    [pscustomobject]@{path=$entry.Name; expected_sha256=$entry.Value.sha256; sha256=$actual; bytes=$size; matches=($actual -eq $entry.Value.sha256 -and $size -eq $entry.Value.bytes)}
}

function Run-ReviewedExecutable {
    param([string]$Name, [string]$Exe, [string[]]$RunArgs)
    $logPath = Join-Path $reviewDir ($Name+'.log')
    $metaPath = Join-Path $reviewDir ($Name+'.json')
    $beforeHash=(Get-FileHash -LiteralPath $Exe -Algorithm SHA256).Hash.ToLowerInvariant()
    $started=[DateTime]::UtcNow.ToString('o')
    $watch=[Diagnostics.Stopwatch]::StartNew()
    $result = & $Exe @RunArgs 2>&1
    $code=$LASTEXITCODE
    $watch.Stop()
    $finished=[DateTime]::UtcNow.ToString('o')
    $result | Out-String -Width 65536 | Set-Content -LiteralPath $logPath -Encoding utf8
    $afterHash=(Get-FileHash -LiteralPath $Exe -Algorithm SHA256).Hash.ToLowerInvariant()
    [pscustomobject]@{argv=@($Exe)+$RunArgs; cwd=$rootPath; started_utc=$started; finished_utc=$finished; elapsed_seconds=$watch.Elapsed.TotalSeconds; executable_sha256=$beforeHash; executable_sha256_after=$afterHash; exit_code=$code; raw_output=$logPath; tested_input_manifest_sha256=(Get-FileHash -LiteralPath $inputPath -Algorithm SHA256).Hash.ToLowerInvariant()} | ConvertTo-Json -Depth 8 | Set-Content -LiteralPath $metaPath -Encoding utf8
    Write-Output ($Name+': exit='+$code+'; duration='+$watch.Elapsed.TotalSeconds+'s; hash_stable='+($beforeHash -eq $afterHash))
    $result | Select-Object -Last 5 | Write-Output
    if ($code -ne 0 -or $beforeHash -ne $afterHash) { throw 'Independent test or binary stability check failed.' }
}
