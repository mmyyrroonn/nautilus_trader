"""Hash actual changed source inputs against the issue's fetched main base."""

import hashlib
import json
import subprocess
import sys
from datetime import UTC, datetime
from pathlib import Path

root = Path("E:/persarb/worktrees/entropy-account")
base = "511ff91d04983afb706eb20989be0b400e6299b9"
paths = subprocess.check_output(["git", "diff", base, "--name-only", "-z"], cwd=root).decode().split("\0")
paths += subprocess.check_output(["git", "ls-files", "--others", "--exclude-standard", "-z"], cwd=root).decode().split("\0")
inputs = {}
for name in sorted(set(paths)):
    if not name or name.startswith("docs/"):
        continue
    path = root / name
    if path.is_file():
        inputs[name] = {"bytes": path.stat().st_size, "sha256": hashlib.sha256(path.read_bytes()).hexdigest()}
record = {"schema_version": 1, "captured_at_utc": datetime.now(UTC).isoformat(),
          "comparison_base": base,
          "candidate_commit": subprocess.check_output(["git", "rev-parse", "HEAD"], cwd=root).decode().strip(),
          "scope": "Changed implementation, tests and generated inputs; credentials excluded", "inputs": inputs}
dest = Path(__file__).parent / (sys.argv[1] if len(sys.argv) > 1 else "tested-inputs.json")
dest.write_text(json.dumps(record, indent=2) + "\n", encoding="utf-8")
print(f"Captured {len(inputs)} actual inputs to {dest.name}")
